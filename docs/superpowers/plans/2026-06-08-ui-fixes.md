## Task 1: Sheet komponenti (drag'siz)

**Files:** Create: `src/shared/components/ui/sheet.tsx`

- [ ] **Step 1: Sheet yaratish**

`src/shared/components/ui/sheet.tsx`:
```tsx
import * as React from 'react';
import { Dialog } from 'radix-ui';
import { cn } from '@shared/lib/utils';

type Side = 'bottom' | 'left' | 'right';

const sideClasses: Record<Side, string> = {
  bottom:
    'inset-x-0 bottom-0 max-h-[85vh] rounded-t-xl border-t data-[state=open]:slide-in-from-bottom data-[state=closed]:slide-out-to-bottom',
  right:
    'inset-y-0 right-0 h-full w-3/4 sm:max-w-sm rounded-l-xl border-l data-[state=open]:slide-in-from-right data-[state=closed]:slide-out-to-right',
  left:
    'inset-y-0 left-0 h-full w-3/4 sm:max-w-sm rounded-r-xl border-r data-[state=open]:slide-in-from-left data-[state=closed]:slide-out-to-left',
};

function Sheet(props: React.ComponentProps<typeof Dialog.Root>) {
  return <Dialog.Root {...props} />;
}

function SheetContent({
  side = 'bottom',
  className,
  children,
  title,
  ...props
}: React.ComponentProps<typeof Dialog.Content> & { side?: Side; title?: string }) {
  return (
    <Dialog.Portal>
      <Dialog.Overlay className="fixed inset-0 z-50 bg-black/10 supports-backdrop-filter:backdrop-blur-xs data-[state=open]:animate-in data-[state=open]:fade-in-0 data-[state=closed]:animate-out data-[state=closed]:fade-out-0" />
      <Dialog.Content
        className={cn(
          'fixed z-50 flex flex-col bg-popover text-sm text-popover-foreground shadow-lg data-[state=open]:animate-in data-[state=closed]:animate-out',
          sideClasses[side],
          className,
        )}
        {...props}
      >
        {/* radix a11y: Title shart; ko'rsatilmasa sr-only */}
        <Dialog.Title className="sr-only">{title ?? 'Panel'}</Dialog.Title>
        {children}
      </Dialog.Content>
    </Dialog.Portal>
  );
}

const SheetClose = Dialog.Close;
const SheetTitle = Dialog.Title;

export { Sheet, SheetContent, SheetClose, SheetTitle };
```

> `slide-in-from-*`/`fade-in-0`/`animate-in` — `tw-animate-css` (index.css'da import qilingan) utilitilari. Agar nom farq qilsa, mavjud `drawer.tsx`'dagi klasslarga qarab moslang.

- [ ] **Step 3: ResizeHandles komponenti**

`src/renderer/components/resize-handles.tsx`:
```tsx
import { ResizeDirection } from '@tauri-apps/api/window';
import { winApi } from '@shared/tauri/window';

const HANDLES: { dir: ResizeDirection; className: string; cursor: string }[] = [
  { dir: ResizeDirection.North, className: 'top-0 inset-x-0 h-1', cursor: 'ns-resize' },
  { dir: ResizeDirection.South, className: 'bottom-0 inset-x-0 h-1', cursor: 'ns-resize' },
  { dir: ResizeDirection.West, className: 'left-0 inset-y-0 w-1', cursor: 'ew-resize' },
  { dir: ResizeDirection.East, className: 'right-0 inset-y-0 w-1', cursor: 'ew-resize' },
  { dir: ResizeDirection.NorthWest, className: 'top-0 left-0 size-2.5', cursor: 'nwse-resize' },
  { dir: ResizeDirection.NorthEast, className: 'top-0 right-0 size-2.5', cursor: 'nesw-resize' },
  { dir: ResizeDirection.SouthWest, className: 'bottom-0 left-0 size-2.5', cursor: 'nesw-resize' },
  { dir: ResizeDirection.SouthEast, className: 'bottom-0 right-0 size-2.5', cursor: 'nwse-resize' },
];

export function ResizeHandles() {
  return (
    <>
      {HANDLES.map(h => (
        <div
          key={h.dir}
          className={`fixed z-40 ${h.className}`}
          style={{ cursor: h.cursor }}
          onPointerDown={e => {
            if (e.button !== 0) return;
            winApi.startResizeDragging(h.dir).catch(() => {});
          }}
        />
      ))}
    </>
  );
}
```
> `ResizeDirection` enum a'zo nomlari (North/South/East/West/NorthEast...) Tauri v2'da shunday; xato bo'lsa context7'dan tasdiqla.
